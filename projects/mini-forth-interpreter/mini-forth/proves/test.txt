>>> from forth import interpret



# COMPROVEM QUE EL CONTINGUT DE LA PILA ES MOSTRA CORRECTAMENT, INCLÒS QUAN 
# ESTÀ BUIDA.

>>> interpret('1 2 3 .s')
[1, 2, 3]

>>> interpret('.s')
[]



# COMPROVEM QUE EL PUNT DESEMPILA EL CIM I EL MOSTRA; I EN CAS QUE NO HI HAGI 
# CAP ELEMENT, ES REPORTI L'ERROR A L'HORA D'INTENTAR DESEMPILAR UNA PILA BUIDA.

>>> interpret('1 2 3 . .s')
3
[1, 2]

>>> interpret('1 . .')
1
Error: pila buida!

# COMPROVACIÓ DEL ÚS DE FUNCIONS


>>> data = '\n'.join([
... ': doble 2 * ;',
... '3 doble .',
... ': f doble 1 + ;',
... '3 f .'
... ])

>>> interpret(data)
6
7


# Si la funcio no existeix, no es calcula res. Sintacticament es correcte, pero 
# no te cap significat degut a la inexistencia de la seva definicio.

>>> interpret('3 f .')
3



# COMPROVACIÓ QUE ELS CONDICIONALS FUNCIONEN CORRECTAMENT


>>> data = '\n'.join([
... ': abs dup 0 < if 0 swap - endif ;',
... '-2 abs .',
... ])

>>> interpret(data)
2

>>> data = '\n'.join([
... ': min 2dup < if drop else swap drop endif ;',
... '3 2 min .',
... ])

>>> interpret(data)
2

>>> data = '\n'.join([
... ': signe dup 0 < if drop -1 else 0 > if 1 else 0 endif endif ;',
... '-2 signe .s',
... ])

>>> interpret(data)
[-1]

>>> data = '\n'.join([
... ': signe dup 0 < if drop -1 else 0 > if 1 else 0 endif endif ;',
... '0 signe .s',
... ])

>>> interpret(data)
[0]



# COMPROVACIÓ QUE LA RECURSIVITAT FUNCIONA CORRECTAMENT

>>> data = '\n'.join([
... ': faux dup rot * swap 1 - dup 2 < if drop else recurse endif ;',
... ': factorial dup 2 < if drop 1 else dup 1 - faux endif ;',
... '4 factorial .',
... '0 factorial .',
... ])

>>> interpret(data)
24
1

# COMPROVACIÓ QUE ELS ERRORS SINTÀCTICS ES DETECTEN I REPORTEN

# Condicionals definits fora d'una funcio. En l'exemple hi ha 2 errors 
# sintàctics: 'if' i 'endif'. Fixar-se que aquestes paraules no es tokenitzen 
# com a identificadors de funcions (tot i que els seus caràcters son valids per 
# ser identificadors), ja que son paraules reservades que es tokenitzen de 
# forma implicita (i els tokens implícits tenen prioritat respecte als 
# explícits).

>>> interpret('1 2 < if .s endif')
2 errors de sintaxi.



# Definicio de funcio incompleta (falta ';')

>>> interpret(': doble 2 * ')
1 errors de sintaxi.



# 'endif' sense 'if'.

>>> data = '\n'.join([
... ': t 1 endif ;',
... 't',
... ])
>>> interpret(data)
1 errors de sintaxi.



# Comentari sense tancar. Aqui nomes es reporta un unic error sintactic: el 
# caràcter '('. Fixar-se que la resta de paraules son valides per ser 
# identificadors de funcions, per això son correctes sintacticament (tot i que 
# semanticament no tenen cap comportament, tal com s'ha demostrat en un altre 
# joc de proves). 

>>> interpret('1 ( comentari sense tancar 2 3 .s')
1 errors de sintaxi.
